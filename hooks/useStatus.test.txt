import { describe, it, expect, vi, beforeEach } from "vitest";
import { renderHook, waitFor } from "@testing-library/react";
import { account } from "@/lib/appwrite";
import { checkUserStatus } from "@/functions/user/check-user-status";
import { useStatus } from "./useStatus";

vi.mock("@/lib/appwrite", () => ({
  account: {
    get: vi.fn(),
  },
}));

vi.mock("@/functions/user/check-user-status", () => ({
  checkUserStatus: vi.fn(),
}));

describe("useStatus", () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  it("deve retornar o status do usuário", async () => {
    (account.get as any).mockResolvedValue({
      $id: "user-123",
    });

    (checkUserStatus as any).mockResolvedValue({
      status: "active",
      isAdmin: false,
    });

    const { result } = renderHook(() => useStatus());

    await waitFor(() => {
      expect(result.current.userStatus).toBe("active");
    });

    expect(account.get).toHaveBeenCalledTimes(1);
    expect(checkUserStatus).toHaveBeenCalledWith("user-123");
  });

  it("deve iniciar com status vazio", () => {
    (account.get as any).mockImplementation(() => new Promise(() => {}));

    const { result } = renderHook(() => useStatus());

    expect(result.current.userStatus).toBe("");
  });

  it("deve manter status vazio quando checkUserStatus falhar", async () => {
    const consoleSpy = vi.spyOn(console, "error").mockImplementation(() => {});

    (account.get as any).mockResolvedValue({
      $id: "user-123",
    });

    (checkUserStatus as any).mockRejectedValue(
      new Error("Erro ao consultar status"),
    );

    const { result } = renderHook(() => useStatus());

    await waitFor(() => {
      expect(checkUserStatus).toHaveBeenCalled();
    });

    expect(result.current.userStatus).toBe("");

    expect(consoleSpy).toHaveBeenCalledWith(
      "Erro ao atualizar ocorrências:",
      expect.any(Error),
    );

    consoleSpy.mockRestore();
  });

  it("deve manter status vazio quando account.get falhar", async () => {
    const consoleSpy = vi.spyOn(console, "error").mockImplementation(() => {});

    (account.get as any).mockRejectedValue(new Error("Erro ao buscar usuário"));

    const { result } = renderHook(() => useStatus());

    await waitFor(() => {
      expect(account.get).toHaveBeenCalled();
    });

    expect(result.current.userStatus).toBe("");

    expect(consoleSpy).toHaveBeenCalledWith(
      "Erro ao atualizar ocorrências:",
      expect.any(Error),
    );

    consoleSpy.mockRestore();
  });
});
